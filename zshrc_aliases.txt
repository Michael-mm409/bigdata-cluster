# --- CSC6002 CLUSTER ALIASES ---
alias lab-push="~/Big-Data-Cluster/infra/sync_labs.sh --push"
alias lab-pull="~/Big-Data-Cluster/infra/sync_labs.sh --pull"
alias hdfs="docker exec -i namenode hdfs"
alias hadoop="docker exec -i namenode hadoop"
alias pyspark="docker exec -it spark-master pyspark"

mongosh() {
    # 1. Safely extract the URI string (keeping it for reference/logging)
    local env_uri=$(awk -F'"' '/^MONGO_URI=/ {print $2}' ~/Big-Data-Cluster/.env)
    
    # 2. Use docker exec to run the shell inside the existing container
    # This assumes your container is named 'course-mongodb' as per your compose file
    if [ -n "$1" ]; then
        # If a db name is provided as an argument
        docker exec -it course-mongodb mongosh "$1"
    else
        # Default connection
        docker exec -it course-mongodb mongosh
    fi
}

cluster-up() {
    docker compose -f ~/Big-Data-Cluster/docker-compose.yml up -d
    ~/Big-Data-Cluster/infra/sync_labs.sh --pull
}

cluster-down() {
    ~/Big-Data-Cluster/infra/sync_labs.sh --push
    docker compose -f ~/Big-Data-Cluster/infra/docker-compose.yml down
}
