# --- CSC6002 CLUSTER ALIASES ---
alias lab-push="~/Big-Data-Cluster/infra/sync_labs.sh --push"
alias lab-pull="~/Big-Data-Cluster/infra/sync_labs.sh --pull"
alias hdfs="docker exec -i namenode hdfs"
alias hadoop="docker exec -i namenode hadoop"
alias pyspark="docker exec -it spark-master pyspark"
alias spark-submit="docker exec -it spark-master spark-submit"

mongosh() {
    # 1. Safely extract the URI string inside the quotes using awk
    local env_uri=$(awk -F'"' '/^MONGO_URI=/ {print $2}' ~/Big-Data-Cluster/.env)

    # 2. Fallback if the .env file line isn't populated or commented out
    local base_uri="${env_uri:-mongodb://localhost:27017/uni_sandbox}"

    # 3. Allow passing a custom database name as an argument to override the trailing path
    if [ -n "$1" ]; then
        base_uri=$(echo "$base_uri" | sed -E "s|/[a-zA-Z0-9__-]+(\?.*)?$|/$1\1|")
    fi

    echo "🔌 Connecting to: $base_uri"
    command mongosh "$base_uri"
}

cluster-up() {
    docker compose -f ~/Big-Data-Cluster/infra/docker-compose.yml up -d
    ~/Big-Data-Cluster/infra/sync_labs.sh --pull
}

cluster-down() {
    ~/Big-Data-Cluster/infra/sync_labs.sh --push
    docker compose -f ~/Big-Data-Cluster/infra/docker-compose.yml down
}
