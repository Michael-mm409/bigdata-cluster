# --- CSC6002 CLUSTER ALIASES ---
alias lab-push="~/Big-Data-Cluster/infra/sync_labs.sh --push"
alias lab-pull="~/Big-Data-Cluster/infra/sync_labs.sh --pull"
alias hadoop="docker exec -i namenode hadoop"
alias pyspark="docker exec -it spark-master pyspark"

unalias hdfs 2>/dev/null

hdfs() {
    docker exec -i namenode hdfs "$@"
}

hdfs-put() {
    local filename="$1"
    local dest="$2"

    if [[ -z "$filename" ]]; then
        echo "Usage: hdfs-put <filename> [hdfs-destination]"
        return 1
    fi

    [[ -z "$dest" ]] && dest="/" || dest="/${dest#/}"

    local source="/course/${filename#/}"
    local source_uri="file://${source// /%20}"

    docker exec -i namenode \
        hdfs dfs -put "$source_uri" "$dest"
}

mongosh() {
    if [[ -n "$1" ]]; then
        docker exec -it course-mongodb mongosh "$1"
    else
        docker exec -it course-mongodb mongosh
    fi
}

cluster-up() {
    docker compose \
        -f ~/Big-Data-Cluster/docker-compose.yml \
        --profile hadoop up -d --scale spark-worker=2

    ~/Big-Data-Cluster/infra/sync_labs.sh --pull
}

cluster-down() {
    ~/Big-Data-Cluster/infra/sync_labs.sh --push

    docker compose \
        -f ~/Big-Data-Cluster/docker-compose.yml \
        --profile hadoop down
}
